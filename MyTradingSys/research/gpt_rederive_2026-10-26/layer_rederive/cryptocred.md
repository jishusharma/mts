## Reproducible audit script

Save this block to an untracked scratch file and run it from the repo root. It reads only the aggregate JSON and Markdown table; it imports no producer calculation module. The assertions passed on 2026-10-28.

```python
import hashlib
import json
from pathlib import Path

ROOT = Path.cwd()
BASE = ROOT / 'MyTradingSys/research/cryptocred_stop_2026-10-10'


def recount():
    rows = json.loads((BASE / 'stop_sweep_result.json').read_text())
    table = []
    for number, line in enumerate((BASE / 'STOP_SOURCES.md').read_text().splitlines(), 1):
        cells = [c.strip() for c in line.strip('|').split('|')]
        if len(cells) == 6 and cells[0].startswith('setup') and cells[1].isdigit():
            table.append((number, cells))
    indexed = {(r['setup'], r['rule']): r for r in rows}
    comparisons = []
    for line, (setup, count, rule, origin, mean, hit) in table:
        r = indexed[setup, rule]
        comparisons.append({'line': line, 'setup': setup, 'rule': rule,
                            'n': r['n'], 'implied_wins': round(r['n'] * r['hit']),
                            'hit_integer_residual': abs(r['n'] * r['hit'] - round(r['n'] * r['hit'])),
                            'net_r': r['net_r'], 'hit': r['hit'],
                            'display_match': (int(count) == r['n'] and origin == r['src']
                                              and f"{r['net_r']:.3f}" == mean and f"{r['hit']:.3f}" == hit)})
    best = {setup: max((r for r in rows if r['setup'] == setup), key=lambda r: r['net_r'])
            for setup in sorted({r['setup'] for r in rows})}
    return {'sha256': hashlib.sha256((BASE / 'stop_sweep_result.json').read_bytes()).hexdigest(),
            'json_rows': len(rows), 'table_rows': len(table),
            'display_mismatches': sum(not r['display_match'] for r in comparisons),
            'nonnegative_net_r_rows': sum(r['net_r'] >= 0 for r in rows),
            'best_per_setup_from_published_aggregates': best,
            'comparisons': comparisons}


if __name__ == '__main__':
    result = recount()
    assert result['json_rows'] == result['table_rows'] == 14
    assert result['display_mismatches'] == result['nonnegative_net_r_rows'] == 0
    assert max(r['hit_integer_residual'] for r in result['comparisons']) < 1e-9
    assert result['best_per_setup_from_published_aggregates']['setup1_sr_flip']['rule'] == 'atr_1.5'
    print(json.dumps(result, indent=2))
```
