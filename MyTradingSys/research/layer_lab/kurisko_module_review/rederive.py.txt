"""Independent synthetic fixture arithmetic from FORMULAS 24; no market data."""
import csv
import json
import math
import re
from fractions import Fraction as F
from pathlib import Path

ROOT=Path.cwd()
BASE=ROOT/'MyTradingSys/backtest/layers/kurisko'
text=(ROOT/'MyTradingSys/backtest/layers/FORMULAS.md').read_text(encoding='utf-8')
example=text.split('**24.2 Worked example',1)[1].split('Bar 4:',1)[0]
ohlc=[tuple(F(x) for x in group) for group in re.findall(r'\((\d+),(\d+),(\d+)\)',example)]
assert len(ohlc)==20
raw=[None]*4
for i in range(4,20):
    window=ohlc[i-4:i+1]
    hh,ll=max(b[0] for b in window),min(b[1] for b in window)
    raw.append(None if hh==ll else 100*(ohlc[i][2]-ll)/(hh-ll))
line=[None]*6+[sum(raw[i-2:i+1])/3 for i in range(6,20)]
count=0
with (BASE/'golden_stoch_k5d3.csv').open(newline='') as f:
    golden=list(csv.DictReader(f))
assert len(golden)==20
for i,r in enumerate(golden):
    assert tuple(F(r[k]) for k in ('high','low','close'))==ohlc[i]
    for key,values in [('raw_k',raw),('line',line)]:
        if values[i] is None:
            assert r[key]==''
        else:
            assert abs(F(r[key])-values[i])<=F(1,10**9)
            count+=1
assert count==30
# Only after deriving every expected value, call the observation module on
# synthetic bars to check gap/NaN restart and prefix causality.
import sys
sys.path.insert(0,str(BASE))
import quad_rotation as Q
TF=60000
bars=[(i*TF,102-i,100-i,100-i) for i in range(90)]
bars += [(90*TF,15,10,14),(91*TF,16,11,15)]
base=Q.compute(bars,TF)
prefixes=0
for n in (20,60,69,85,90,91):
    got=Q.compute(bars[:n],TF)
    for a,b in zip(got,base[:n]):
        assert a[:5]==b[:5] and a.episode_start_t==b.episode_start_t
        assert all(x==y or math.isnan(x) and math.isnan(y) for x,y in zip(a.values,b.values))
    prefixes+=1
gap=[b if i<85 else (b[0]+TF,*b[1:]) for i,b in enumerate(bars)]
nan=bars[:85]+[(bars[85][0],math.nan,math.nan,math.nan)]+bars[86:]
for series in (gap,nan):
    got=Q.compute(series,TF)
    assert all(r.side is None for r in got[85:])
    assert all(math.isnan(v) for r in got[85:] for v in r.values)
print(json.dumps(dict(golden_rows=20,finite_values_rederived=count,blank_warmup_values=10,
                     prefix_checks=prefixes,gap_nan_restarts=2,
                     first_D_exact=str(line[6]),first_D_float=float(line[6]))))
