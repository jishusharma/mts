"""Post-freeze implementation comparison. Does not import or read implementation tests."""
import hashlib
import importlib.util
import json
import pathlib
import runpy

ROOT = pathlib.Path(__file__).resolve().parents[2]
HOME = pathlib.Path(__file__).resolve().parent


def load(name):
    path = ROOT / 'backtest' / 'bt' / (name + '.py')
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module


def capture(fn, *args):
    try:
        return fn(*args)
    except Exception as exc:
        return dict(error=type(exc).__name__, message=str(exc))


def compare():
    frozen_path = HOME / 'gpt_formulas_rederive_10_frozen.json'
    frozen = json.loads(frozen_path.read_text())
    oracle = runpy.run_path(str(HOME / 'gpt_formulas_rederive_10_independent.py.txt'))
    assert oracle['derive']() == frozen, 'Frozen expected values changed'
    bt, ref = load('bt_stats'), load('ref_pbo')
    x, variance = frozen['returns'], float(oracle['moments'](oracle['TRIAL_SR'])[1][0])
    # Oracle uses four variant rows by eight time columns. APIs use time rows.
    api_matrix = list(map(list, zip(*frozen['matrix'])))
    a, b = bt.pbo_cscv(api_matrix, 4), ref.ref_pbo_cscv(api_matrix, 4)
    comparisons = []

    def record(name, expected, actual):
        comparisons.append(dict(metric=name, expected=expected, actual=actual,
                                abs_difference=abs(expected - actual)))

    record('bt sharpe', frozen['sharpe'], bt.sharpe(x))
    record('ref sharpe', frozen['sharpe'], ref.sharpe(x))
    record('bt psr', frozen['psr']['probability'], bt.psr(x))
    record('bt expected_max_sr', frozen['expected_max_sr'], bt.expected_max_sr(variance, 5))
    record('bt dsr', frozen['dsr']['probability'], bt.dsr(x, 5, variance))
    record('bt pbo', frozen['pbo']['pbo'], a[0])
    record('ref pbo', frozen['pbo']['pbo'], b[0])
    for i, split in enumerate(frozen['pbo']['splits']):
        record(f'bt logit {i}', split['logit'], a[1][i])
        record(f'ref logit {i}', split['logit'], b[1][i])
        record(f'ref rank w {i}', split['w'], b[2][i])
    edges = {}
    for label, xs in [('empty', []), ('singleton', [1]), ('constant_positive', [1]*10),
                      ('constant_negative', [-1]*10), ('all_zero', [0]*10)]:
        edges[label] = dict(bt_sharpe=capture(bt.sharpe, xs), ref_sharpe=capture(ref.sharpe, xs),
                            bt_psr=capture(bt.psr, xs), bt_dsr=capture(bt.dsr, xs, 5, variance))
    edges['expected_max'] = {str((v,n)): capture(bt.expected_max_sr,v,n)
                             for v,n in [(variance,1),(0,5),(-1,5),(variance,0)]}
    tied = [[1]*4, [-1]*4]*4
    cases = {'all_tied': (tied,4), 'short_T': (api_matrix[:2],4),
             'zero_S': (api_matrix,0), 'negative_S': (api_matrix,-2), 'odd_S': (api_matrix,3),
             'empty_M': ([],4), 'no_variants': ([[]]*8,4),
             'ragged_M': (api_matrix[:3]+[[1]]+api_matrix[4:],4),
             'trailing_row': (api_matrix+[[100,-100,50,-50]],4)}
    for label,(matrix,s) in cases.items():
        edges[label] = dict(bt=capture(bt.pbo_cscv,matrix,s),ref=capture(ref.ref_pbo_cscv,matrix,s))
    record('psr at own SR', .5, bt.psr(x,bt.sharpe(x)))
    record('positive scale Sharpe',frozen['sharpe'],bt.sharpe([v*7 for v in x]))
    record('positive scale PSR',frozen['psr']['probability'],bt.psr([v*7 for v in x]))
    # A nonzero skew supplement checks the term not activated by the blind symmetric fixture.
    skewed = [*oracle['RETURNS'][:-1], oracle['Fraction'](9,100)]
    mean,(v,m3,m4) = oracle['moments'](skewed)
    sr = oracle['ratio'](skewed)
    skew,kurt = float(m3)/float(v)**1.5,float(m4/v**2)
    record('supplement skewed PSR',oracle['probability'](sr,0,10,skew,kurt)['probability'],
           bt.psr(list(map(float,skewed))))
    result = dict(comparisons=comparisons,max_abs_difference=max(r['abs_difference'] for r in comparisons),
                  edges=edges, source_sha256={name: hashlib.sha256((ROOT/'backtest'/'bt'/(name+'.py')).read_bytes()).hexdigest()
                                            for name in ['bt_stats','ref_pbo']})
    assert result['max_abs_difference'] < 1e-12
    return result


if __name__ == '__main__':
    result = compare()
    (HOME/'gpt_formulas_rederive_10_comparison.json').write_text(json.dumps(result,indent=2)+'\n',encoding='utf-8',newline='\n')
    print(json.dumps(result,indent=2))
