print(f'At n=200: SE={se_n200:.6f}; MDE above cost={mde_n200:.6f}; gross threshold={.14+mde_n200:.6f}; threshold/per-encounter sd={(.14+mde_n200)/sd_t:.6f} SD')
print(f'At n=40, true mean=C, chance observed mean exceeds C+2SE = {1-NormalDist().cdf(2):.6%}')
print('PASS_FLOW: n_T=40, mean_T=-.0568, C=.14, t_cost=-1.51, DIFF=-.0500, t_DIFF=-.37 => fails')
print('PASS_CONTROL: n_C=1439, mean_C=-.0081, C=.14, t_cost=-6.03 => fails')

# Gate / trial arithmetic, following power_mde.py's documented two-point payoff.
p, rr, cost_r, n = .35, 4, 0.0, 30
mean_net = p*(rr-cost_r)+(1-p)*(-1-cost_r)
var_net = p*((rr-cost_r)-mean_net)**2+(1-p)*((-1-cost_r)-mean_net)**2
mde_r = (z975+z80)*sqrt(var_net)/sqrt(n)
print('PRE-AUDIT VARIANT ARITHMETIC')
print(f'1-.975**10 = {1-.975**10:.8%}')
