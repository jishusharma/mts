# ---- GPT side: definitions only ----
def g_fwd(sgn, closes, i, h):
    if i + h >= len(closes):
        return None, None
    ret = sgn * (closes[i + h] - closes[i]) / closes[i]
    mae = min(sgn * (closes[j] - closes[i]) / closes[i] for j in range(i + 1, min(i + h, len(closes) - 1) + 1))
    return ret, mae

def g_atr(h, l, c, n):
    tr = [max(h[i] - l[i], abs(h[i] - c[i - 1]), abs(l[i] - c[i - 1])) for i in range(1, len(h))]
    return sum(tr[:n]) / n

def g_net(sgn, pin, pout, tin, tout, recs, fee=0.0010, slip=0.0002):
    gross = sgn * (pout - pin) / pin
    return gross - fee - slip - sgn * sum(r for t, r in recs if tin < t <= tout)

def g_netrr(e, t, s, taker=0.0005):
    if s == e:
        return 0.0
    cost = 2 * taker * e
    return (abs(t - e) - cost) / (abs(e - s) + cost)

rows = []
def add(name, formulas, gpt, repo, line):
    ok = isinstance(repo, (int, float)) and abs(formulas - gpt) < 1e-9 and abs(gpt - repo) < 1e-9
    rows.append((name, formulas, gpt, repo, "MATCH" if ok else ("NOT_IN_DATA" if repo is None else "MISMATCH @ " + line)))
