"""Reproduce the three FORMULAS.md 14-16 event comparisons on a synthetic fixture.
Run from repository root: python MyTradingSys/research/layer_lab/reports/2026-10-28_gpt-formulas-rederive-14-16.py.txt
"""
from datetime import datetime, timezone
from pathlib import Path
import sys

sys.path.insert(0, str(Path(__file__).resolve().parents[4]))

from MyTradingSys.backtest.layers.positioning.funding_cvd_fade import funding_cvd_fade_events
from MyTradingSys.backtest.layers.session.session_momentum import session_momentum_events
from MyTradingSys.backtest.layers.turn_of_candle.turn_of_candle import turn_of_candle_events

N = 4320
MIN = 60_000
DAY = 1440
START = int(datetime(2020, 1, 1, tzinfo=timezone.utc).timestamp() * 1000)

# Spec-only fixture. Create intended cumulative CVD path, then encode its increments
# using the module API's delta = 2*tbv - v fields.
closes = [100.0 + i * 0.001 for i in range(N)]
for day_start in (0, DAY, 2 * DAY):
    closes[day_start + 29] = 101.0 if day_start == 0 else 100.0 + (day_start + 29) * 0.001
funding_indices = list(range(480, 3841, 480))
for i in funding_indices:
    closes[i] = max(closes[i - 2], closes[i - 1]) + 0.5

cvd = [float(i) for i in range(N)]
for i in funding_indices:
    cvd[i] = float(i - 3)  # strictly below both prior CVD observations

deltas = [cvd[0]] + [cvd[i] - cvd[i - 1] for i in range(1, N)]
bars = []
for i in range(N):
    delta = deltas[i]
    if delta < 0:
        tbv, volume = 0.0, -delta
    else:
        tbv, volume = delta, delta
    t = START + i * MIN
    o = 100.0 if i in (0, DAY, 2 * DAY) else closes[i]
    bars.append({"t": t, "o": o, "c": closes[i], "tbv": tbv, "v": volume})

funding = [(START + i * MIN, i / 1000.0) for i in range(0, 3841, 480)]

actual = {
    "funding_fade": [e["i"] for e in funding_cvd_fade_events(
        bars, funding, lookback=3, n=2, hi=2 / 3, lo=1 / 3, use_divergence=True
    )],
    "session_momentum": [e["i"] for e in session_momentum_events(
        bars, bar_min=1, session_start_min=0, first_min=30,
        entry_min=30, min_abs=0.0
    )],
    "turn_of_candle": [e["i"] for e in turn_of_candle_events(
        bars, lead_minutes=1, side="short"
    )],
}
expected = {
    "funding_fade": list(range(960, 3841, 480)),
    "session_momentum": [29, 1469, 2909],
    "turn_of_candle": list(range(14, N, 15)),
}
for name in expected:
    got = actual[name]
    want = expected[name]
    print(f"{name}: expected_count={len(want)} actual_count={len(got)}")
    print(f"  expected_indices={want if name != 'turn_of_candle' else '14 + 15*k (k=0..287)'}")
    print(f"  actual_indices={got if name != 'turn_of_candle' else '14 + 15*k (k=0..287)'}")
    print(f"  match={got == want}")
