  def window(body, pos):
      a, b = max(0, pos - 90), pos + 170
      s = body[a:b]
      s = s.split(' ', 1)[1] if a > 0 and ' ' in s else s
      s = s.rsplit(' ', 1)[0] if ' ' in s else s
