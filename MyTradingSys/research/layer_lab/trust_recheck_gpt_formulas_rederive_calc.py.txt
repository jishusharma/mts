# Independent stdlib arithmetic, written from FORMULAS.md sections 13 and 17 before code inspection.
entry, target, stop = 100.0, 104.0, 98.0
taker = 0.0005
cost = 2 * taker * entry
net_rr = (abs(target - entry) - cost) / (abs(entry - stop) + cost)
gross = (101.0 - 100.0) / 100.0
fee_rt, slip_rt = 0.0010, 0.0002
funding_sum, side_sign = 0.0001 + 0.0003, 1
s13_net = gross - fee_rt - slip_rt - side_sign * funding_sum
print(f'net_rr={net_rr:.15f}')
print(f's13_long_net={s13_net:.15f}')
