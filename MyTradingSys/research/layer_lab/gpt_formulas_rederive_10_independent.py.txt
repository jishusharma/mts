import itertools
import json
import math
from fractions import Fraction
from statistics import NormalDist

RETURNS = [Fraction(x, 100) for x in [-3, -1, 0, 1, 2, 4, -2, 3, 1, 5]]
TRIAL_SR = [Fraction(x, 10) for x in [-4, -1, 2, 5, 8]]
# Four strategy rows, eight time columns. Four contiguous blocks of two times.
MATRIX = [[3, 2, -2, -1, 1, 4, -3, 0],
          [-1, 2, 4, 3, -2, 0, 1, -3],
          [1, -2, 0, 3, 4, -1, 2, 1],
          [0, 1, 2, -3, 3, 2, -1, 4]]
GAMMA = 0.5772156649015329
NORMAL = NormalDist()


def moments(xs):
    mean = sum(xs) / len(xs)
    return mean, [sum((x - mean) ** p for x in xs) / len(xs) for p in (2, 3, 4)]


def ratio(xs):
    mean, (v, _, _) = moments(xs)
    return float(mean) / math.sqrt(float(v))


def probability(sr, benchmark, n, skew, kurt):
    denominator_squared = 1 - skew * sr + (kurt - 1) * sr * sr / 4
    z = (sr - benchmark) * math.sqrt(n - 1) / math.sqrt(denominator_squared)
    return {"denominator_squared": denominator_squared, "z": z, "probability": NORMAL.cdf(z)}


def maximum(variance, n):
    return math.sqrt(variance) * ((1 - GAMMA) * NORMAL.inv_cdf(1 - 1 / n)
                                  + GAMMA * NORMAL.inv_cdf(1 - 1 / (n * math.e)))


def cscv(matrix):
    rows = []
    for blocks in itertools.combinations(range(4), 2):
        ins = [t for t in range(8) if t // 2 in blocks]
        outs = [t for t in range(8) if t // 2 not in blocks]
        train = [ratio([Fraction(s[t]) for t in ins]) for s in matrix]
        test = [ratio([Fraction(s[t]) for t in outs]) for s in matrix]
        winner = max(range(4), key=lambda j: train[j])
        rank = 1 + sum(v < test[winner] for v in test)
        w = rank / 5
        logit = math.log(w / (1 - w))
        rows.append(dict(blocks=list(blocks), winner=winner, train=train, test=test,
                         rank=rank, w=w, logit=logit, overfit=logit <= 0))
    return dict(splits=rows, pbo=sum(row["overfit"] for row in rows) / len(rows))


def derive():
    mean, (v, m3, m4) = moments(RETURNS)
    sr = ratio(RETURNS)
    skew, kurt = float(m3) / float(v) ** 1.5, float(m4 / v ** 2)
    trial_mean, (trial_var, _, _) = moments(TRIAL_SR)
    threshold = maximum(float(trial_var), 5)
    return dict(returns=list(map(float, RETURNS)), trial_sr=list(map(float, TRIAL_SR)), matrix=MATRIX,
                mean=str(mean), population_variance=str(v), central_m3=str(m3), central_m4=str(m4),
                sharpe=sr, skew=skew, kurtosis=kurt, trial_mean=str(trial_mean),
                trial_variance=str(trial_var), expected_max_sr=threshold,
                psr=probability(sr, 0, 10, skew, kurt),
                dsr=probability(sr, threshold, 10, skew, kurt), pbo=cscv(MATRIX),
                edges={"all_tied_nonconstant_rows_pbo": cscv([[1, -1] * 4] * 4)["pbo"],
                       "constant_returns": "undefined: zero volatility",
                       "empty_returns": "undefined: no moments",
                       "one_return": "undefined: zero volatility and T-1=0",
                       "one_trial": "zero expected max under centered null; approximation singular",
                       "zero_trial_variance_n5": 0.0,
                       "negative_trial_variance": "undefined: negative variance",
                       "psr_at_own_sr": probability(sr, sr, 10, skew, kurt)["probability"]})


if __name__ == "__main__":
    print(json.dumps(derive(), indent=2, allow_nan=False))
