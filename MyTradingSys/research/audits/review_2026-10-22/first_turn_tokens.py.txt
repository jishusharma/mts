import json,datetime,statistics
from pathlib import Path
root=Path.home()/".claude"/"projects"
rows=[]
for f in root.glob("*/*.jsonl"):
    if f.stat().st_mtime<datetime.datetime(2026,10,12).timestamp() or "scratch-2026-10-22" in str(f): continue
    for line in f.open(encoding="utf-8",errors="ignore"):
        try:e=json.loads(line)
        except:continue
        u=(e.get("message") or {}).get("usage")
        if e.get("type")=="assistant" and u and not e.get("isSidechain"):
            tot=u.get("input_tokens",0)+u.get("cache_read_input_tokens",0)+u.get("cache_creation_input_tokens",0)
            if tot>1000:
                rows.append((e["timestamp"][:10],tot,u.get("input_tokens",0)));break
import collections
d=collections.defaultdict(list)
for day,t,i in rows: d[day].append(t)
for day in sorted(d): v=d[day]; print(day,"sessions",len(v),"median first-turn full prompt",int(statistics.median(v)),"min",min(v),"max",max(v))
